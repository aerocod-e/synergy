PATCH /api/trips/:tripId/start
Protected: authenticateToken
Allowed roles: driver only
Logic:
  Find trip by tripId where deleted_at IS NULL
  Validate:
    - Trip must exist, return 404 if not
    - req.user.userId must match trip.driver_id
      return 403 "You are not assigned to this trip"
    - Trip status must be 'created'
      return 409 "Trip cannot be started from 
      its current status"
  
  Update trips:
    status = 'in_transit'
    updated_at = NOW()
  
  Insert trip_events:
    event_type = 'started'
    performed_by = req.user.userId
    metadata = { previousStatus: 'created' }

PATCH /api/trips/:tripId/cancel
Protected: authenticateToken
Allowed roles: super_admin only
Logic:
  Find trip by tripId where deleted_at IS NULL
  Validate:
    - Trip must exist, return 404 if not
    - Trip status must NOT be 'cancelled' 
      or 'completed'
      return 409 "Trip cannot be cancelled from 
      its current status"

  Run everything below in a single transaction:

  STEP A: Reverse all wallet credits for this trip
    Find all wallet_ledger entries where:
      trip_id = tripId
      entry_type = 'credit'
    
    For each ledger entry:
      1. INSERT new wallet_ledger row:
           wallet_id = same wallet_id
           trip_id = same trip_id
           trip_fee_id = same trip_fee_id
           amount = original amount as NEGATIVE integer
           entry_type = 'reversal'
           cleared = false
      
      2. UPDATE wallets:
           total_collected = total_collected - 
           original amount
           updated_at = NOW()

  STEP B: Soft delete the trip
    UPDATE trips:
      status = 'cancelled'
      deleted_at = NOW()
      updated_at = NOW()

  STEP C: Insert trip_events:
    event_type = 'cancelled'
    performed_by = req.user.userId
    metadata = { 
      previousStatus: <status before cancel>,
      reversedEntries: <count of ledger reversals>
    }

  On any failure: rollback entire transaction

Response 200:
{
  "data": {
    "tripId": "...",
    "status": "cancelled",
    "reversedEntries": 0,
    "walletsRestored": 0
  },
  "message": "Trip cancelled and wallets reversed"
}
