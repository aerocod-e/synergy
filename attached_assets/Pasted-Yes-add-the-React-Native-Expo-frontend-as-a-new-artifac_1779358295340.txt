   export type UserRole = 
     'super_admin' | 'admin' | 'driver'

     'state' | 'stakeholder' | null

     id: string

     accessToken: string
     refreshToken: string

     'created' | 'in_transit' | 
     'completed' | 'cancelled'

     'loading' | 'unloading' | 
     'stakeholder_fee'

   export type RecipientType = 
     'state' | 'stakeholder'

   export interface Trip {
     id: string
     vehicleNumber: string
     status: TripStatus
     loadingState: { id: string; name: string; code: string }
     unloadingState: { id: string; name: string; code: string }
     driver: { id: string; email: string }
     originDescription: string
     destinationDescription: string
     unstartedFlag: boolean | null
     createdAt: string
     updatedAt: string
   }

   export interface WalletLedgerEntry {
     ledgerId: string
     tripId: string
     vehicleNumber: string
     feeType: FeeType
     amount: number
     amountNaira: string
     entryType: 'credit' | 'reversal'
     cleared: boolean
     clearedAt: string | null
     createdAt: string
   }

5. Create services/api.ts:
   A base API client with these functions:

   - Store accessToken and refreshToken 
     in memory (not AsyncStorage yet, 
     we will add persistence in the 
     auth step)

   - async function apiRequest(
       path: string,
       options?: RequestInit,
       requiresAuth = true
     ): Promise<any>
     
     Builds full URL from API_BASE_URL + path
     Adds Content-Type: application/json header
     If requiresAuth adds Authorization: 
       Bearer <accessToken> header
     On 401 response: attempt one token 
       refresh via POST /api/auth/refresh
       If refresh succeeds: retry original 
         request with new token
       If refresh fails: clear tokens and 
         throw error with code 'AUTH_EXPIRED'
     Returns response.data from parsed JSON
     Throws on non-2xx responses with the 
       error and message from response body

   - Export named functions that wrap 
     apiRequest for each endpoint:

     auth:
       login(email, password)
       logout(refreshToken)
       getMe()

     trips:
       getTrips(params?)
       getTripById(tripId)
       createTrip(body)
       startTrip(tripId)
       completeTrip(tripId)
       cancelTrip(tripId)

     wallets:
       getMyWallet()
       getMyLedger(params?)
       getAllWallets()

     lookup:
       lookupVehicle(vehicleNumber)
