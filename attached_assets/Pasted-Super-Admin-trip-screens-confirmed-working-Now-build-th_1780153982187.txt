NAIRA FORMATTING HELPER
Create utils/formatCurrency.ts:

export function formatNaira(
  kobo: number
): string {
  const naira = kobo / 100
  return naira.toLocaleString('en-NG', {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  })
}

export function formatNairaWithSign(
  kobo: number, 
  entryType: 'credit' | 'reversal'
): string {
  const formatted = formatNaira(
    Math.abs(kobo)
  )
  return entryType === 'credit' 
    ? '+₦' + formatted 
    : '-₦' + formatted
}

Use this utility in BOTH the wallet 
screen and ledger screen and in any 
future screens that display money.
