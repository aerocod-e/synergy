sql-- For each vehicle type, insert rates for all three stakeholders
-- Use subqueries to look up IDs by name rather than hardcoding UUIDs
-- Wrap in DO $$ block for idempotency

DO $$
DECLARE
  v_chart_id UUID;
  v_consultant_id UUID;
  v_vendor_id UUID;
  v_integrator_id UUID;
BEGIN
  SELECT id INTO v_chart_id FROM pricing_charts WHERE is_active = true LIMIT 1;
  SELECT id INTO v_consultant_id FROM stakeholders WHERE name = 'Consultant';
  SELECT id INTO v_vendor_id FROM stakeholders WHERE name = 'Vendor';
  SELECT id INTO v_integrator_id FROM stakeholders WHERE name = 'Integrator';

  IF v_chart_id IS NULL THEN RETURN; END IF;

  -- Delete existing rates for this chart to re-seed cleanly
  DELETE FROM pricing_chart_stakeholder_rates WHERE pricing_chart_id = v_chart_id;

  -- Insert rates: (chart, stakeholder, vehicle_type, amount)
  INSERT INTO pricing_chart_stakeholder_rates (pricing_chart_id, stakeholder_id, vehicle_type_id, amount)
  SELECT v_chart_id, v_consultant_id, id,
    CASE name
      WHEN 'Buses and Equivalent' THEN 15000
      WHEN 'Pickup Vans and Equivalent (4 Tyres)' THEN 0
      WHEN 'Pickup Non-Heavy Duty 6/8 Tyres' THEN 15000
      WHEN 'Heavy Duty 6/8 Tyres' THEN 40000
      WHEN 'Luxury Buses and Equivalent' THEN 50000
      WHEN '12/14 Tyres and Equivalent' THEN 220000
      WHEN '18 Tyres and above' THEN 350000
    END
  FROM vehicle_types;

  INSERT INTO pricing_chart_stakeholder_rates (pricing_chart_id, stakeholder_id, vehicle_type_id, amount)
  SELECT v_chart_id, v_vendor_id, id,
    CASE name
      WHEN 'Buses and Equivalent' THEN 15000
      WHEN 'Pickup Vans and Equivalent (4 Tyres)' THEN 50000
      WHEN 'Pickup Non-Heavy Duty 6/8 Tyres' THEN 50000
      WHEN 'Heavy Duty 6/8 Tyres' THEN 50000
      WHEN 'Luxury Buses and Equivalent' THEN 50000
      WHEN '12/14 Tyres and Equivalent' THEN 150000
      WHEN '18 Tyres and above' THEN 200000
    END
  FROM vehicle_types;

  INSERT INTO pricing_chart_stakeholder_rates (pricing_chart_id, stakeholder_id, vehicle_type_id, amount)
  SELECT v_chart_id, v_integrator_id, id,
    CASE name
      WHEN 'Buses and Equivalent' THEN 15000
      WHEN 'Pickup Vans and Equivalent (4 Tyres)' THEN 70000
      WHEN 'Pickup Non-Heavy Duty 6/8 Tyres' THEN 100000
      WHEN 'Heavy Duty 6/8 Tyres' THEN 150000
      WHEN 'Luxury Buses and Equivalent' THEN 200000
      WHEN '12/14 Tyres and Equivalent' THEN 200000
      WHEN '18 Tyres and above' THEN 200000
    END
  FROM vehicle_types;
END $$;
